\subsection{Type-0 Neighborhood}

\paragraph{Type-0 Elimination}
%%
The center $X$ of the neighborhood is such that:
%%
\begin{itemize}
\item $X$ is non-root, non-hypothesis, and non-collapsed;
\item $X$ is the conclusion of $⊃$E;
\item $X$'s incoming and outgoing d-edges have color 0;
\item $X$ has no incoming a-edges;
\item $X$ has no parents with an incoming a-edge (i.e., ``no a-edges up'').
\end{itemize}
%%
\leanfile[firstline=137,lastline=179]{hc.lean}

\paragraph{Type-0 Introduction}
%%
The center $X$ of the neighborhood is such that:
%%
\begin{itemize}
\item $X$ is non-root, non-hypothesis, and non-collapsed;
\item $X$ is the conclusion of $⊃$I;
\item $X$'s incoming and outgoing d-edges have color 0;
\item $X$ has no incoming a-edges or a-edges up.
\end{itemize}
%%
\leanfile[firstline=181,lastline=213]{hc.lean}

\paragraph{Type-0 Hypothesis}
%%
The center $X$ of the neighborhood is such that:
%%
\begin{itemize}
\item $X$ is non-root and non-collapsed;
\item $X$ is a hypothesis;
\item $X$ has no incoming d-edges;
\item $X$'s single outgoing d-edge has color 0;
\item $X$ has no incoming a-edges or a-edges up.
\end{itemize}
%%
\leanfile[firstline=215,lastline=238]{hc.lean}
